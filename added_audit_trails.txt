Now here are the Vultr update commands:

  # 1. SSH into the server
  ssh -i ssh_vultr root@<your-vultr-ip>

  # 2. Go to the app directory
  cd /var/www/faktura_scanner_flask   # adjust path if different

  # 3. Pull latest changes
  git fetch origin
  git checkout feat/ui-improvements-todos   # or your deployed branch
  git pull origin feat/ui-improvements-todos

  # 4. Activate virtualenv and install any new dependencies
  source .venv/bin/activate
  pip install -r requirements.txt

  # 5. Run DB schema migration (safe — all statements use IF NOT EXISTS / IF EXISTS)
  python -c "from config.database import initialize_database; initialize_database()"

  # 6. Rebuild TailwindCSS if templates changed
  npm run build:css

  # 7. Restart gunicorn
  sudo systemctl restart gunicorn
  # — or, if using supervisor —
  sudo supervisorctl restart faktura_scanner

  # 8. Check the app is healthy
  sudo systemctl status gunicorn
  curl -I http://localhost:5000/
